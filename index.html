async function updateHeaderTitle() { let headerTitle = document.getElementById('main-header-title'); if (!headerTitle) return; let currentLang = localStorage.getItem('pos_current_lang') || 'lo'; let t = translations[currentLang] || translations['lo']; let userRole = localStorage.getItem('pos_user_role'); if (userRole === 'admin') { headerTitle.innerHTML = `🛒 ລະບົບຂາຍເຄື່ອງ POS ອັດສະລິຍະ ລະດັບສູງສຸດ — Super Admin Control Center [ Full Management & Realtime Sync System ]`; return; } // ດຶງຄ່າຈາກ LocalStorage let activeUsername = localStorage.getItem('pos_active_shop_username') || ''; let isShopOwner = localStorage.getItem('pos_shop_owner') === 'true'; let permanentDevId = localStorage.getItem('pos_permanent_device_id') || localStorage.getItem('pos_hardware_locked_id') || deviceId; let shopName = ''; // ຖ້າເປັນເຈົ້າຂອງຮ້ານ (ລັອກອິນດ້ວຍ Username) ໃຫ້ດຶງຊື່ຮ້ານຈາກ Username ນັ້ນ if (activeUsername) { shopName = localStorage.getItem('pos_shop_name_' + activeUsername) || localStorage.getItem('pos_shop_name_' + deviceId); } else { shopName = localStorage.getItem('pos_shop_name_' + deviceId); } let displayName = ""; // 📌 ເງື່ອນໄຂຕັດສິນ: ຖ້າເປັນຮ້ານເປີດຖາວອນ (isShopOwner ຫຼື activeUsername ມີຄ່າ) ແລະ ມີຊື່ຮ້ານແລ້ວ if ((isShopOwner || activeUsername !== '') && shopName && shopName.trim() !== '' && shopName.trim().toUpperCase() !== activeUsername.toUpperCase() && !shopName.toUpperCase().startsWith('POS-')) { displayName = shopName; // ສະແດງສະເພາະຊື່ຮ້ານ } else { // 📌 ຖ້າເປັນລະບົບທົດລອງ 15 ວັນ ໃຫ້ສະແດງສະເພາະ Hardware/Permanent ID (POS-DEV-...) ຢ່າງດຽວເທົ່ານັ້ນ displayName = permanentDevId; } headerTitle.innerHTML = `🛒 ${t.title}   [ ${displayName} ]`; }
